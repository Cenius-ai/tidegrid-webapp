=== Tidegrid demo ===

$ tidegrid --version
tidegrid 1.0.0

$ tidegrid --help
usage: tidegrid [-h] [--version] COMMAND ...

Edit tidal harmonic constants and predict high and low waters.

options:
  -h, --help    show this help message and exit
  --version     show program's version number and exit

commands:
  COMMAND
    init        create a station file
    show        print the station summary
    validate    check the station file
    set         edit station-level fields
    constituents
                list the built-in constituent catalog
    list        list the station's constituents
    add         append a harmonic constant
    remove      delete a harmonic constant
    predict     predict height or high/low waters
    export      write a tide table

examples:
  tidegrid init --file station.json --name 'Port Haven' --datum MLLW --z0 2.35
  tidegrid add M2 --file station.json --amplitude 3.72 --phase 118.4
  tidegrid show --file station.json
  tidegrid predict --file station.json --date 2024-06-01
  tidegrid export --file station.json --from 2024-06-01 --to 2024-06-03 --format csv --out tides.csv

exit codes: 0 success, 1 usage error, 2 data or validation error

=== exit-code contract ===

$ tidegrid frobnicate   # expect exit 1
stderr: usage: tidegrid [-h] [--version] COMMAND ...
stderr: tidegrid: error: argument COMMAND: invalid choice: 'frobnicate' (choose from 'init', 'show', 'validate', 'set', 'constituents', 'list', 'add', 'remove', 'predict', 'export')
exit: 1

$ tidegrid set --file station.json --units fathoms   # expect exit 1
stderr: usage: tidegrid set [-h] [--file PATH] [--no-color] [--z0 M] [--name NAME]
stderr:                     [--datum DATUM] [--units {m,ft}] [--tz-offset MINUTES]
stderr: tidegrid set: error: argument --units: invalid choice: 'fathoms' (choose from 'm', 'ft')
exit: 1

$ tidegrid show --file missing.json   # expect exit 2
stderr: tidegrid: station file 'missing.json' not found - create it with: tidegrid init --file missing.json
exit: 2

$ tidegrid predict --file station.json --date 2024-13-45   # expect exit 2
stderr: tidegrid: '2024-13-45' is not a real calendar date - use YYYY-MM-DD, for example 2024-06-01
exit: 2

$ tidegrid add NOTREAL --file station.json --amplitude 0.5 --phase 10   # expect exit 2
stderr: tidegrid: NOTREAL: unknown constituent - run 'tidegrid constituents' for the catalog
exit: 2

$ tidegrid add M2 --file station.json --amplitude 0.10 --phase 5   # expect exit 2
stderr: tidegrid: M2: already present in station.json - remove it first with: tidegrid remove M2 --file station.json
exit: 2

$ tidegrid remove M6 --file station.json   # expect exit 2
stderr: tidegrid: M6: not present in station.json - run 'tidegrid list --file station.json' to see the stored constituents
exit: 2
